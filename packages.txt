# Packages ShowPlayPI installs on top of Raspberry Pi OS Lite (2026-06-18, trixie, arm64).
# Installed without recommends; dependencies are pulled in by apt automatically and are intentionally
# not listed here.

# --- Graphical kiosk session ---
chromium
xserver-xorg-core
xserver-xorg-legacy
xserver-xorg-input-libinput
xserver-xorg-video-all
xinit
x11-xserver-utils
dbus-x11
openbox
unclutter-xfixes
fonts-dejavu-core

# --- Remote control / VNC ---
x11vnc
xdotool

# --- Python libraries used by the ShowPlayPI services ---
# IDNA 2008 host names like Chromium (showplaypi_url.py)
python3-idna

# --- Idle timeout (showplaypi-idle reads the X idle time via libXss) ---
libxss1

# --- Tools for configuration and the USB gadget ---
ipcalc
fatattr
mc

# --- Boot splash ---
rpi-splash-screen-support
imagemagick
# Network share of the SHOWPLAYPI drive; wsdd2 shows it in the Windows network view
samba
wsdd2

# --- Video mode (showplaypi-video) ---
mpv

# --- Sound: one output that plays on every HDMI port, the headphone jack (Pi 4) and USB sound cards ---
# PipeWire runs for the user admin from boot (rootfs/var/lib/systemd/linger/admin)
pipewire
wireplumber
pipewire-pulse
pipewire-alsa
# Real-time priority for the audio threads (no dropouts under high CPU load)
rtkit

# --- Ontime mode: Ontime runs as a container (no permanently running daemon, unlike Docker) ---
podman
